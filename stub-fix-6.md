## Issue Title
bedrock-alpha: grant methods limit to only one region

## Generated At
2026-01-29T16:20:35.044Z

## Analysis Summary
The CrossRegionInferenceProfile's grant methods only provide permissions for the current region, but cross-region inference profiles require access to all regions within the specified geoRegion (EU, US, APAC).

## Root Cause
The grantProfileUsage() method in CrossRegionInferenceProfile class only creates IAM permissions for the current region, while the service needs permissions across multiple regions within the same geographical area.

## Proposed Changes
### Change 1: modify packages/@aws-cdk/aws-bedrock-alpha/lib/inference-profiles/cross-region-inference-profile.ts
Update CrossRegionInferenceProfile to grant permissions across all relevant regions within a geoRegion
```typescript
import { RegionInfo } from 'aws-cdk-lib/region-info';

private getRegionsForGeoArea(): string[] {
  const prefixMappings: Record<CrossRegionInferenceProfileRegion, string[]> = {
    [CrossRegionInferenceProfileRegion.EU]: ['eu-'],
    [CrossRegionInferenceProfileRegion.US]: ['us-'],
    [CrossRegionInferenceProfileRegion.APAC]: ['ap-'],
  };
  
  const prefixes = prefixMappings[this.geoRegion];
  if (!prefixes) {
    return ['*'];
  }
  
  return RegionInfo.regions
    .filter(r => r.partition === 'aws')
    .map(r => r.name)
    .filter(name => prefixes.some(prefix => name.startsWith(prefix)));
}

public grantProfileUsage(grantee: IGrantable): Grant {
  const resourceArns = this.getRegionsForGeoArea().map(region => 
    Arn.format({
      partition: Aws.PARTITION,
      service: 'bedrock',
      account: Aws.ACCOUNT_ID,
      region: region,
      resource: 'inference-profile',
      resourceName: this.inferenceProfileId,
      arnFormat: ArnFormat.SLASH_RESOURCE_NAME,
    })
  );
  
  return Grant.addToPrincipal({
    grantee: grantee,
    actions: ['bedrock:GetInferenceProfile', 'bedrock:InvokeModel*'],
    resourceArns: resourceArns,
  });
}
```
